1. Trim basico

derivated column

id = trim(id)
nombre = trim(nombre)

2. Filtro para nulos o vacios

filter

!isNull(id) &&
!isNull(nombre) &&
length(id) > 0 &&
length(nombre) > 0

3. Normalizar texto a minusculas

derivated column

id = lower(id)
nombre =lower(nombre)

4. Convertir id a numero

derivated column

id = toInteger(regexReplace(id, '[^0-9]', ''))

5. Corregir nombre

derivated column

nombre = regexReplace(replace(nombre, '-', ' '),'[^a-zA-Z0-9\\s]','')

6. Eliminar duplicados segun nombre

aggregate

group by nombre as nombre
id = first(id)
